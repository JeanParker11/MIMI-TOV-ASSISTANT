## 🔧 SYSTÈME TECHNIQUE

### **Structure JSON Robuste**
```json
{
  "competence_id": {
    "id": "string (unique, snake_case)",
    "nom": "string (français)",
    "rang": "E|D|C|B|A|S",
    "type": "active|passive",
    "categorie": "offensif|defensif|mixte",
    "faction": {
      "allowed": ["array"],
      "excluded": ["array"]
    },
    "utilisations_par_rang": "number",
    "taux_reussite": "number (20-90%)",
    "description": "string (détaillée)",
    "mecanique": {
      "declencheur": "string",
      "condition": "object|null",
      "effet": {
        "type": "string",
        // propriétés spécifiques à l'effet
      },
      "cout": {
        "type": "mana",
        "valeur": "number",
        // ou valeur_par_seconde, valeur_par_minute, etc.
        "cooldown_minutes": "number (optionnel)"
      }
    }
  }
}
```
